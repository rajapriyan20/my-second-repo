    def render_polars_table():
        nonlocal current_limit
        if not current_df_name or current_df_name not in dfs_dict:
            with preview_output:

        t0 = time.time()

            col = f["col_dd"].value
            op = f["op_dd"].value
            val = f["val_txt"].value.strip()
            dtype_str = col_types.get(col, "")

            if op == "is_not_null":
                filter_exprs.append(pl.col(col).is_not_null())
                code_snippets.append(f'pl.col("{col}").is_not_null()')
            elif op == "is_null":
                filter_exprs.append(pl.col(col).is_null())
                code_snippets.append(f'pl.col("{col}").is_null()')
            elif val:
                if op == "contains":
                    filter_exprs.append(pl.col(col).cast(pl.Utf8).str.contains(val, literal=True))
                    code_snippets.append(f'pl.col("{col}").cast(pl.Utf8).str.contains(r"{val}", literal=True)')
                elif op == "starts_with":
                    filter_exprs.append(pl.col(col).cast(pl.Utf8).str.starts_with(val))
                    code_snippets.append(f'pl.col("{col}").cast(pl.Utf8).str.starts_with("{val}")')
                elif op == "ends_with":
                    filter_exprs.append(pl.col(col).cast(pl.Utf8).str.ends_with(val))
                    code_snippets.append(f'pl.col("{col}").cast(pl.Utf8).str.ends_with("{val}")')
                elif op == "==":
                    if "Int" in dtype_str or "Float" in dtype_str:
                        try:
                            num = float(val) if "." in val else int(val)
                            filter_exprs.append(pl.col(col) == num)
                            code_snippets.append(f'pl.col("{col}") == {num}')
                        except Exception:
                            filter_exprs.append(pl.col(col).cast(pl.Utf8) == val)
                            code_snippets.append(f'pl.col("{col}").cast(pl.Utf8) == "{val}"')
                    else:
                        filter_exprs.append(pl.col(col).cast(pl.Utf8) == val)
                        code_snippets.append(f'pl.col("{col}") == "{val}"')
                elif op == "!=":
                    filter_exprs.append(pl.col(col).cast(pl.Utf8) != val)
                    code_snippets.append(f'pl.col("{col}").cast(pl.Utf8) != "{val}"')
                elif op in (">", ">=", "<", "<="):
                    try:
                        num = float(val) if "." in val else int(val)
                        if op == ">": filter_exprs.append(pl.col(col) > num)
                        elif op == ">=": filter_exprs.append(pl.col(col) >= num)
                        elif op == "<": filter_exprs.append(pl.col(col) < num)
                        elif op == "<=": filter_exprs.append(pl.col(col) <= num)
                        code_snippets.append(f'pl.col("{col}") {op} {num}')
                    except Exception:
                        filter_exprs.append(pl.col(col).cast(pl.Utf8) > val)

                    for expr in filter_exprs[1:]:

                    lf_plan = lf_plan.filter(combined_expr)
                
                # Count matching rows lazily

                # Projection and limit pushed down directly to Parquet scanner

            else:
                df_data = obj
                if filter_exprs:
                    combined_expr = filter_exprs[0]
                    for expr in filter_exprs[1:]:
                        combined_expr = combined_expr & expr
                    filtered_df = df_data.filter(combined_expr)
