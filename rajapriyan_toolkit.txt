        function runFilters() {{
            if (!currentDf || !meta[currentDf]) return;
            const dfInfo = meta[currentDf];
            const colsToShow = dfInfo.col_names.filter(c => appliedCols.has(c));
            document.getElementById("visibleColsCount").innerText = colsToShow.length + " visible";

            const filterExprs = [];
            const codeParts = [];

            activeFilters.forEach(f => {{
                const col = f.colSel.value;
                const op = f.opSel.value;
                const val = (f.valInput.value || "").trim();
                const colIdx = dfInfo.col_names.indexOf(col);
                if (colIdx === -1) return;

                if (op === "is_not_null") {{
                    filterExprs.push(row => row[colIdx] !== "null" && row[colIdx] !== null && row[colIdx] !== "");
                    codeParts.push(`pl.col("${{col}}").is_not_null()`);
                }} else if (op === "is_null") {{
                    filterExprs.push(row => row[colIdx] === "null" || row[colIdx] === null || row[colIdx] === "");
                    codeParts.push(`pl.col("${{col}}").is_null()`);
                }} else if (val) {{
                    const valLower = val.toLowerCase();
                    if (op === "contains") {{
                        filterExprs.push(row => String(row[colIdx]).toLowerCase().includes(valLower));
                        codeParts.push(`pl.col("${{col}}").cast(pl.Utf8).str.contains(r"${{val}}", literal=True)`);
                    }} else if (op === "starts_with") {{
                        filterExprs.push(row => String(row[colIdx]).toLowerCase().startsWith(valLower));
                        codeParts.push(`pl.col("${{col}}").cast(pl.Utf8).str.starts_with("${{val}}")`);
                    }} else if (op === "ends_with") {{
                        filterExprs.push(row => String(row[colIdx]).toLowerCase().endsWith(valLower));
                        codeParts.push(`pl.col("${{col}}").cast(pl.Utf8).str.ends_with("${{val}}")`);
                    }} else if (op === "==") {{
                        filterExprs.push(row => String(row[colIdx]).toLowerCase() === valLower);
                        codeParts.push(`pl.col("${{col}}") == "${{val}}"`);
                    }} else if (op === "!=") {{
                        filterExprs.push(row => String(row[colIdx]).toLowerCase() !== valLower);
                        codeParts.push(`pl.col("${{col}}") != "${{val}}"`);
                    }} else if (op === ">" || op === ">=" || op === "<" || op === "<=") {{
                        const num = parseFloat(val);
                        if (!isNaN(num)) {{
                            if (op === ">") filterExprs.push(row => parseFloat(row[colIdx]) > num);
                            if (op === ">=") filterExprs.push(row => parseFloat(row[colIdx]) >= num);
                            if (op === "<") filterExprs.push(row => parseFloat(row[colIdx]) < num);
                            if (op === "<=") filterExprs.push(row => parseFloat(row[colIdx]) <= num);
                            codeParts.push(`pl.col("${{col}}") ${{op}} ${{num}}`);
                        }}
                    }}
                }}
            }});

            // Execute Filter across preview records
            let matchedRows = dfInfo.preview_data;
            if (filterExprs.length > 0) {{
                matchedRows = dfInfo.preview_data.filter(row => filterExprs.every(fn => fn(row)));
            }}

            // Update Polars Code Snippet
            const codeStr = codeParts.length > 0 
                ? `${{currentDf}}.filter(${{codeParts.join(" & ")}})`
                : `${{currentDf}}.head(50)`;
            document.getElementById("codeSnippetText").innerText = "Code: " + codeStr;
            document.getElementById("matchStatsLabel").innerText = `⚡ Matched ${{matchedRows.length}} rows in preview`;

            const tbody = document.getElementById("previewTbody");

            let thHtml = '<tr class="bg-slate-900 text-cyan-400 font-mono text-xs uppercase"><th style="width: 50px; text-align: center; position: relative;">#</th>';
            colsToShow.forEach(c => {{
                thHtml += `<th style="width: 160px; position: relative;" title="${{c}}">${{c}}<div class="col-resizer"></div></th>`;
            }});
            thHtml += '</tr>';
            thead.innerHTML = thHtml;

            let tbHtml = '';
            matchedRows.forEach((row, rIdx) => {{
                const rowClass = (rIdx % 2 === 0) ? "row-even" : "row-odd";
                let cells = `<td style="text-align: center; color: #64748b;">${{rIdx + 1}}</td>`;
                colsToShow.forEach(c => {{
                    const cIdx = dfInfo.col_names.indexOf(c);
                    const val = (cIdx !== -1) ? row[cIdx] : "null";
                    cells += `<td title="${{val}}">${{val}}</td>`;
                }});
                tbHtml += `<tr class="${{rowClass}}">${{cells}}</tr>`;
            }});
