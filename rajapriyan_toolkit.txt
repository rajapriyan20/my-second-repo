    btn_reset_filters.add_class(f"{uid}_btn_secondary")

    status_code_box = widgets.HTML(value="")

    def update_col_checkboxes():
        if not current_df_name or current_df_name not in dfs_dict:
            col_checklist_box.children = []
            return
        
        df_obj = dfs_dict[current_df_name]
        query = col_search_input.value.strip().lower()
        
        checkbox_widgets = []
        for col in df_obj.columns:
            if query and query not in col.lower():
                continue
            
            is_checked = (col in pending_cols)
            chk = widgets.Checkbox(
                value=is_checked,
                description=f"{col} ({df_obj[col].dtype})",
                indent=False,
                layout=widgets.Layout(width='auto', margin='2px 3px')
            )
            chk.add_class(f"{uid}_col_chk_pill")
            if is_checked:
                chk.add_class("is-checked")
            
            def make_chk_handler(c_name, w_chk):
                def handler(change):
                    if change['new']:
                        pending_cols.add(c_name)
                        w_chk.add_class("is-checked")
                    else:
                        pending_cols.discard(c_name)
                        w_chk.remove_class("is-checked")
                return handler

            chk.observe(make_chk_handler(col, chk), names='value')
            checkbox_widgets.append(chk)

        col_checklist_box.children = checkbox_widgets

    def create_filter_row():
        if not current_df_name or current_df_name not in dfs_dict:
            return

            options=[

                ('equals (==)', '=='),
                ('not equals (!=)', '!='),
                ('starts with', 'starts_with'),
                ('ends with', 'ends_with'),
                ('> (greater)', '>'),
                ('>= (greater/eq)', '>='),
                ('< (less)', '<'),
                ('<= (less/eq)', '<='),
                ('is not null', 'is_not_null'),
                ('is null', 'is_null')
            ],
            value='contains',
            layout=widgets.Layout(width='130px')
        )
        val_txt = widgets.Text(placeholder="Value...", layout=widgets.Layout(width='140px'))
        btn_del = widgets.Button(description="✕", layout=widgets.Layout(width='32px'))
        btn_del.add_class(f"{uid}_btn_danger")

        row_box = widgets.HBox([col_dd, op_dd, val_txt, btn_del], layout=widgets.Layout(margin='2px 0', align_items='center'))
        
        filter_item = {
            "widget": row_box,
            "col_dd": col_dd,
            "op_dd": op_dd,
            "val_txt": val_txt
        }
        active_filters.append(filter_item)

        def remove_row(b):
            if filter_item in active_filters:
                active_filters.remove(filter_item)
                filter_rows_container.children = [f["widget"] for f in active_filters]

        btn_del.on_click(remove_row)
        filter_rows_container.children = [f["widget"] for f in active_filters]

    def render_polars_table():
        if not current_df_name or current_df_name not in dfs_dict:
            with preview_output:
                preview_output.clear_output()
                display(HTML("<div style='padding:16px; color:#64748b;'>No active Polars DataFrame selected.</div>"))
            return

        df_obj = dfs_dict[current_df_name]
        t0 = time.time()

        # Build Polars expression dynamically
        filter_exprs = []
        code_snippets = []

        for f in active_filters:
            col = f["col_dd"].value
            op = f["op_dd"].value
            val = f["val_txt"].value.strip()
            dtype_str = str(df_obj[col].dtype)

            if op == "is_not_null":
                filter_exprs.append(pl.col(col).is_not_null())
                code_snippets.append(f'pl.col("{col}").is_not_null()')
            elif op == "is_null":
                filter_exprs.append(pl.col(col).is_null())
                code_snippets.append(f'pl.col("{col}").is_null()')
            elif val:
                if op == "contains":
                    filter_exprs.append(pl.col(col).cast(pl.Utf8).str.contains(val, literal=True))
                    code_snippets.append(f'pl.col("{col}").cast(pl.Utf8).str.contains(r"{val}", literal=True)')
                elif op == "starts_with":
                    filter_exprs.append(pl.col(col).cast(pl.Utf8).str.starts_with(val))
                    code_snippets.append(f'pl.col("{col}").cast(pl.Utf8).str.starts_with("{val}")')
                elif op == "ends_with":
                    filter_exprs.append(pl.col(col).cast(pl.Utf8).str.ends_with(val))
                    code_snippets.append(f'pl.col("{col}").cast(pl.Utf8).str.ends_with("{val}")')
                elif op == "==":
                    if "Int" in dtype_str or "Float" in dtype_str:
                        try:
                            num = float(val) if "." in val else int(val)
                            filter_exprs.append(pl.col(col) == num)
                            code_snippets.append(f'pl.col("{col}") == {num}')
                        except Exception:
                            filter_exprs.append(pl.col(col).cast(pl.Utf8) == val)
                            code_snippets.append(f'pl.col("{col}").cast(pl.Utf8) == "{val}"')
                    else:
                        filter_exprs.append(pl.col(col).cast(pl.Utf8) == val)
                        code_snippets.append(f'pl.col("{col}") == "{val}"')
                elif op == "!=":
                    filter_exprs.append(pl.col(col).cast(pl.Utf8) != val)
                    code_snippets.append(f'pl.col("{col}").cast(pl.Utf8) != "{val}"')
                elif op in (">", ">=", "<", "<="):
                    try:
                        num = float(val) if "." in val else int(val)
                        if op == ">": filter_exprs.append(pl.col(col) > num)
                        elif op == ">=": filter_exprs.append(pl.col(col) >= num)
                        elif op == "<": filter_exprs.append(pl.col(col) < num)
                        elif op == "<=": filter_exprs.append(pl.col(col) <= num)
                        code_snippets.append(f'pl.col("{col}") {op} {num}')
                    except Exception:
                        filter_exprs.append(pl.col(col).cast(pl.Utf8) > val)
                        code_snippets.append(f'pl.col("{col}").cast(pl.Utf8) {op} "{val}"')

        # Execute Polars Filter across memory
        try:
            if filter_exprs:
                combined_expr = filter_exprs[0]
                for expr in filter_exprs[1:]:
                    combined_expr = combined_expr & expr
                filtered_df = df_obj.filter(combined_expr)
            else:
                filtered_df = df_obj
        except Exception as err:
            with preview_output:
                preview_output.clear_output()
                display(HTML(f"<div style='padding:16px; color:#f87171;'>Polars Filter Error: {html.escape(str(err))}</div>"))
            return

        elapsed_ms = (time.time() - t0) * 1000
        matched_count = filtered_df.shape[0]

        # Apply Column Visibility
        cols_to_show = [c for c in df_obj.columns if c in applied_columns]
        if not cols_to_show:
            cols_to_show = df_obj.columns[:8]
            applied_columns.clear()
            applied_columns.update(cols_to_show)

        slice_df = filtered_df.select(cols_to_show).head(preview_rows)

        # Status & Generated Code Info Box
        code_str = f'{current_df_name}.filter({" & ".join(code_snippets)})' if code_snippets else f'{current_df_name}.head({preview_rows})'
        status_html = f"""
        <div style="background:#0f172a; border:1px solid #334155; border-radius:8px; padding:8px 12px; margin:0 0 10px 0;">
            <div style="display:flex; justify-content:space-between; align-items:center; font-size:12px; margin-bottom:4px;">
                <span style="color:#34d399; font-weight:600;">⚡ Matched {matched_count:,} of {df_obj.shape[0]:,} rows <span style="color:#94a3b8; font-weight:normal;">({elapsed_ms:.1f} ms)</span></span>
                <span style="color:#38bdf8; font-family:monospace; font-size:11px;">Showing top {min(preview_rows, matched_count)} rows × {len(cols_to_show)} cols (Click header to sort · Drag edge to resize)</span>
            </div>
            <div style="font-family:monospace; font-size:11px; color:#94a3b8; background:#1e293b; padding:5px 10px; border-radius:6px; overflow-x:auto;">
                <span style="color:#38bdf8;">Code:</span> {html.escape(code_str)}
            </div>
        </div>
        """
        status_code_box.value = status_html

        # Table Geometry with 3-State Sort & Draggable Width Resizing
        table_id = f"{uid}_res_table_el"
        th_cells = f'<th style="width:50px; min-width:50px; max-width:50px; text-align:center; position:relative; cursor:default;" data-no-sort="true">#</th>'
        for col in cols_to_show:
            th_cells += f'<th style="width:150px; min-width:150px; max-width:150px; position:relative;" title="Click to sort (Asc/Desc/Orig) · Drag edge to resize" data-sort="none"><span class="col-title">{html.escape(col)}</span><span class="sort-icon" style="color:#64748b; margin-left:4px; font-size:10px;"></span><div class="{uid}_resizer"></div></th>'
