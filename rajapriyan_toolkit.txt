from IPython.display import Javascript, display

# Triggers a clean frontend restart in both classic Jupyter and JupyterLab/Mosaic
display(Javascript("""
if (window.jupyterapp) {
    window.jupyterapp.commands.execute('kernelmenu:restart');
} else if (window.Jupyter && Jupyter.notebook) {
    Jupyter.notebook.kernel.restart();
}
"""))
