# --- Acoustic Protocol Parameters ---
SAMPLE_RATE = 44100
SYMBOL_DURATION = 0.030  # 30 ms per symbol (~33 baud)
SAMPLES_PER_SYM = int(SAMPLE_RATE * SYMBOL_DURATION)
CHUNK_PAYLOAD_SIZE = 36  # Bytes per chunk (~1.1s audio transmission per chunk)

# 16-FSK Frequencies for Data (Hex nibbles 0x0 - 0xF)
FSK_FREQS = [1600 + i * 150 for i in range(16)]  # 1600 Hz to 3850 Hz

# Control Frequencies
FREQ_PING = 1100  # Device Handshake Ping
FREQ_PONG = 1350  # Device Handshake Pong
FREQ_CHUNK_START = 4400  # Start of a chunk transmission
FREQ_ACK = 5000  # Receiver receipt confirmation beep
FREQ_EOF = 5600  # End of transmission

# Global cancellation and control flags
tx_cancel_evt = threading.Event()
rx_cancel_evt = threading.Event()
pair_stop_evt = threading.Event()

# --- Signal Processing Helpers ---


def make_tone(freq, duration=0.2, amp=0.5):
    """Generates an amplitude-tapered sine wave to prevent speaker clicks."""
    t = np.linspace(0, duration, int(SAMPLE_RATE * duration), endpoint=False)
    win = np.sin(np.pi * np.linspace(0, 1, len(t)))
    return (np.sin(2 * np.pi * freq * t) * win * amp).astype(np.float32)


def make_chirp(start_f, end_f, duration=0.25, amp=0.5):
    """Generates a frequency sweep for synchronization."""
    t = np.linspace(0, duration, int(SAMPLE_RATE * duration), endpoint=False)
    phase = 2 * np.pi * (start_f * t + (end_f - start_f) * (t**2) / (2 * duration))
    win = np.sin(np.pi * np.linspace(0, 1, len(t)))
    return (np.sin(phase) * win * amp).astype(np.float32)


def get_tone_snr(audio_chunk, target_freq, tolerance=60):
    """Measures relative Signal-to-Noise Ratio (SNR) for a given frequency."""
    if len(audio_chunk) < 512:
        return 0.0
    spectrum = np.abs(rfft(audio_chunk))
    freqs = rfftfreq(len(audio_chunk), 1.0 / SAMPLE_RATE)
    mask = (freqs >= target_freq - tolerance) & (freqs <= target_freq + tolerance)
    if not np.any(mask):
        return 0.0
    peak = np.max(spectrum[mask])
    floor = np.median(spectrum) + 1e-6
    return peak / floor


def decode_nibble(audio_block):
    """Identifies the dominant 16-FSK nibble frequency."""
    spectrum = np.abs(rfft(audio_block))
    freqs = rfftfreq(len(audio_block), 1.0 / SAMPLE_RATE)
    floor = np.median(spectrum) + 1e-6

    max_snr = 0.0
    best_nibble = None
    for nibble, target in enumerate(FSK_FREQS):
        mask = (freqs >= target - 45) & (freqs <= target + 45)
        if np.any(mask):
            snr = np.max(spectrum[mask]) / floor
            if snr > max_snr and snr > 2.8:
                max_snr = snr
                best_nibble = nibble
    return best_nibble


def encode_data_chunk(chunk_idx, total_chunks, data_bytes):
    """Encodes a single chunk with header into audio waveform."""
    # Chunk packet structure: [2B index] + [2B total] + [1B len] + [payload] + [1B checksum]
    chk = sum(data_bytes) & 0xFF
    header = struct.pack(">HHB", chunk_idx, total_chunks, len(data_bytes))
    packet = header + data_bytes + bytes([chk])

    tones = []
    # 1. Preamble Chirp
    tones.append(make_chirp(2200, FREQ_CHUNK_START, duration=0.15, amp=0.6))
    tones.append(np.zeros(int(SAMPLE_RATE * 0.03), dtype=np.float32))

    # 2. FSK Data Tones
    t_sym = np.linspace(
        0, SYMBOL_DURATION, SAMPLES_PER_SYM, endpoint=False
    ).astype(np.float32)
    win = np.sin(np.pi * np.linspace(0, 1, SAMPLES_PER_SYM)).astype(np.float32)

    for b in packet:
        for nibble in [(b >> 4) & 0x0F, b & 0x0F]:
            f = FSK_FREQS[nibble]
            tone = (np.sin(2 * np.pi * f * t_sym) * win * 0.5).astype(
                np.float32
            )
            tones.append(tone)

    return np.concatenate(tones)

def on_test_spk(_):

    wave = np.concatenate([make_tone(n, 0.12, 0.4) for n in notes])
    sd.play(wave, samplerate=SAMPLE_RATE, blocking=False)


def on_test_mic(_):
    btn_test_mic.disabled = True

    def worker():
        with tab1_out:
            clear_output()
            print("🎙️ Listening... Speak, snap fingers, or tap near laptop.")
        levels = []

        def cb(indata, frames, t_info, status):
            lvl = float(np.max(np.abs(indata[:, 0])))
            levels.append(lvl)
            vu_meter.value = min(lvl * 3.5, 1.0)

        with sd.InputStream(
            channels=1,
            samplerate=SAMPLE_RATE,
            blocksize=2048,
            callback=cb,
        ):
            time.sleep(5.0)

        vu_meter.value = 0
        btn_test_mic.disabled = False
        with tab1_out:
            clear_output()
            avg = np.mean(levels) if levels else 0
            if avg > 0.015:
                print(f"✅ Mic OK! Average input level: {avg * 100:.1f}%")
            else:
                print(
                    "⚠️ Low mic input detected. Increase microphone volume in OS settings."
                )

    threading.Thread(target=worker).start()

def execute_pairing(_):
    pair_stop_evt.clear()
    btn_pair_action.disabled = True
    role = pair_role.value

    if "Initiator" in role:

        def init_worker():
            with tab2_out:
                clear_output()
                print("🔊 Sending Handshake Ping (1100 Hz)...")
            pair_status_banner.value = (
                "<div style='color:#0275d8;'><b>Status:</b> Ping sent! Waiting for Laptop B echo...</div>"
            )

            # Play Ping
            sd.play(
                make_tone(FREQ_PING, 0.35, 0.6),
                samplerate=SAMPLE_RATE,
                blocking=True,
            )

            heard_pong = False
            t0 = time.time()

            def cb(indata, frames, t_info, status):
                nonlocal heard_pong
                if pair_stop_evt.is_set():
                    raise sd.CallbackStop
                if get_tone_snr(indata[:, 0], FREQ_PONG) > 3.8:
                    heard_pong = True

            with sd.InputStream(
                channels=1,
                samplerate=SAMPLE_RATE,
                blocksize=2048,
                callback=cb,
            ):
                while (time.time() - t0 < 6.0) and not heard_pong:
                    if pair_stop_evt.is_set():
                        break
                    time.sleep(0.05)

            btn_pair_action.disabled = False
            with tab2_out:
                clear_output()
                if heard_pong:
                    pair_status_banner.value = (
                        "<div style='color:#5cb85c; font-weight:bold;'>✅ Status: PAIRED! Two-way audio confirmed.</div>"
                    )
                    print(
                        "🎉 Both devices can clearly hear each other. You are ready to transfer files!"
                    )
                else:
                    pair_status_banner.value = (
                        "<div style='color:#d9534f;'>❌ Status: No response from Laptop B.</div>"
                    )
                    print(
                        "Make sure Laptop B has 'Laptop B (Responder)' selected and clicked 'Start Pairing Test'."
                    )

        threading.Thread(target=init_worker).start()

    else:  # Responder

        def resp_worker():
            with tab2_out:
                clear_output()
                print(
                    "👂 Listening for Laptop A's ping... (Press Stop to cancel)"
                )
            pair_status_banner.value = (
                "<div style='color:#0275d8;'><b>Status:</b> Listening for Laptop A...</div>"
            )

            heard_ping = False

            def cb(indata, frames, t_info, status):
                nonlocal heard_ping
                if pair_stop_evt.is_set():
                    raise sd.CallbackStop
                if get_tone_snr(indata[:, 0], FREQ_PING) > 3.8:
                    heard_ping = True

            with sd.InputStream(
                channels=1,
                samplerate=SAMPLE_RATE,
                blocksize=2048,
                callback=cb,
            ):
                while not heard_ping and not pair_stop_evt.is_set():
                    time.sleep(0.05)

            if heard_ping:
                time.sleep(0.1)
                # Reply with Pong
                sd.play(
                    make_tone(FREQ_PONG, 0.35, 0.6),
                    samplerate=SAMPLE_RATE,
                    blocking=True,
                )
                pair_status_banner.value = (
                    "<div style='color:#5cb85c; font-weight:bold;'>✅ Status: Heard Ping & Sent Pong! Paired.</div>"
                )
                with tab2_out:
                    print("⚡ Ping received! Returned Pong response.")
            btn_pair_action.disabled = False

        threading.Thread(target=resp_worker).start()

# --- SENDER LOGIC (Chunk -> Play -> Wait for ACK -> Repeat) ---


def run_sender(_):
    tx_cancel_evt.clear()
    path = txt_send_file.value.strip()

    if not os.path.exists(path):
        with out_sender:
            clear_output()
            print(f"❌ File not found: '{path}'")
        return

    btn_send_start.disabled = True
    btn_send_cancel.disabled = False

    with open(path, "rb") as f:
        raw_bytes = f.read()

    file_hash = hashlib.sha256(raw_bytes).hexdigest()
    compressed = gzip.compress(raw_bytes)

    # Slice into payload chunks
    chunks = [
        compressed[i : i + CHUNK_PAYLOAD_SIZE]
        for i in range(0, len(compressed), CHUNK_PAYLOAD_SIZE)
    ]
    total_chunks = len(chunks)

    send_progress.max = total_chunks
    send_progress.value = 0

    with out_sender:
        clear_output()
        print(f"📦 File: {os.path.basename(path)}")
        print(
            f"📊 Size: {len(raw_bytes):,} bytes ➔ Compressed: {len(compressed):,} bytes"
        )
        print(
            f"🧩 Sliced into {total_chunks} chunks (~{(total_chunks * 1.5):.0f}s total with confirmations)"
        )
        print(f"🔑 SHA-256 (Prefix): {file_hash[:16]}")
        print("-" * 50)

    def tx_worker():
        for idx, chk_data in enumerate(chunks):
            if tx_cancel_evt.is_set():
                break

            sender_status.value = (
                f"<b>Status:</b> 🔊 Sending chunk {idx + 1}/{total_chunks}..."
            )
            chunk_wave = encode_data_chunk(idx, total_chunks, chk_data)

            # Transmit Chunk Audio
            sd.play(chunk_wave, samplerate=SAMPLE_RATE, blocking=True)

            if tx_cancel_evt.is_set():
                break

            # Listen for Receiver ACK beep (1.5s timeout)
            sender_status.value = f"<b>Status:</b> 📡 Waiting for Receiver ACK ({idx + 1}/{total_chunks})..."
            ack_received = False
            t_listen_start = time.time()

            def ack_cb(indata, frames, t_info, status):
                nonlocal ack_received
                if tx_cancel_evt.is_set():
                    raise sd.CallbackStop
                if get_tone_snr(indata[:, 0], FREQ_ACK) > 3.5:
                    ack_received = True

            with sd.InputStream(
                channels=1,
                samplerate=SAMPLE_RATE,
                blocksize=1024,
                callback=ack_cb,
            ):
                while (time.time() - t_listen_start < 1.8) and not ack_received:
                    if tx_cancel_evt.is_set():
                        break
                    time.sleep(0.02)

            send_progress.value = idx + 1
            with out_sender:
                if ack_received:
                    print(
                        f" Chunk {idx + 1:02d}/{total_chunks:02d} transmitted ➔  Received confirmation beep"
                    )
                else:
                    print(
                        f" Chunk {idx + 1:02d}/{total_chunks:02d} transmitted (⚠️ ACK missed, advancing...)"
                    )

            time.sleep(0.05)  # Brief pause between chunks

        btn_send_start.disabled = False
        btn_send_cancel.disabled = True

        if tx_cancel_evt.is_set():
            sender_status.value = (
                "<b style='color:#d9534f;'>Status: Transmission CANCELLED by user.</b>"
            )
            with out_sender:
                print("\n🛑 Sending aborted.")
        else:
            # Send End of File tone
            sd.play(
                make_tone(FREQ_EOF, 0.3, 0.5),
                samplerate=SAMPLE_RATE,
                blocking=True,
            )
            sender_status.value = (
                "<b style='color:#5cb85c;'>Status: Transmission COMPLETE!</b>"
            )
            with out_sender:
                print("\n✨ All chunks successfully transmitted!")

    threading.Thread(target=tx_worker).start()


def cancel_sender(_):
    tx_cancel_evt.set()
    sd.stop()
    btn_send_start.disabled = False
    btn_send_cancel.disabled = True


btn_send_start.on_click(run_sender)
btn_send_cancel.on_click(cancel_sender)

# --- RECEIVER LOGIC (Detect Chunk -> Decode -> Send ACK Beep) ---


def run_receiver(_):
    rx_cancel_evt.clear()
    btn_recv_start.disabled = True
    btn_recv_cancel.disabled = False
    recv_progress.value = 0

    with out_receiver:
        clear_output()
        print(
            "🎧 Receiver active! Listening for incoming audio chunks... (Press Cancel to stop)"
        )
    receiver_status.value = (
        "<b>Status:</b> 🎧 Listening for audio stream..."
    )

    def rx_worker():
        buffer = []
        received_chunks = {}
        total_expected_chunks = None
        state = "SEARCH_CHUNK"

        def audio_cb(indata, frames, t_info, status):
            if rx_cancel_evt.is_set():
                raise sd.CallbackStop
            buffer.extend(indata[:, 0])

        with sd.InputStream(
            channels=1,
            samplerate=SAMPLE_RATE,
            blocksize=1024,
            callback=audio_cb,
        ):
            while not rx_cancel_evt.is_set():
                # State 1: Search for chunk start chirp
                if state == "SEARCH_CHUNK":
                    if len(buffer) >= 4096:
                        chunk_sample = np.array(buffer[:4096])
                        if (
                            get_tone_snr(
                                chunk_sample, FREQ_CHUNK_START, tolerance=100
                            )
                            > 3.5
                        ):
                            state = "DECODE_CHUNK"
                            # Fast forward past chirp
                            buffer = buffer[
                                int(
                                    4096
                                    + SAMPLE_RATE * (SYMBOL_DURATION + 0.03)
                                ) :
                            ]
                        else:
                            buffer = buffer[2048:]

                # State 2: Decode nibbles for this chunk
                elif state == "DECODE_CHUNK":
                    nibbles = []
                    # Read enough symbols for header (5 bytes = 10 nibbles) + max payload
                    needed_samples = (
                        5 + CHUNK_PAYLOAD_SIZE + 1
                    ) * 2 * SAMPLES_PER_SYM

                    timeout_t = time.time()
                    while (
                        len(buffer) < needed_samples
                        and not rx_cancel_evt.is_set()
                    ):
                        if time.time() - timeout_t > 2.0:  # Timeout safety
                            break
                        time.sleep(0.01)

                    if rx_cancel_evt.is_set():
                        break

                    # Decode all available symbols in the chunk
                    while len(buffer) >= SAMPLES_PER_SYM:
                        sym = np.array(buffer[:SAMPLES_PER_SYM])
                        buffer = buffer[SAMPLES_PER_SYM:]
                        nib = decode_nibble(sym)
                        if nib is not None:
                            nibbles.append(nib)

                    # Reconstruct bytes from nibbles
                    pkt_bytes = bytearray()
                    for i in range(0, len(nibbles) - 1, 2):
                        pkt_bytes.append((nibbles[i] << 4) | nibbles[i + 1])

                    # Verify header and checksum
                    if len(pkt_bytes) >= 6:
                        c_idx, c_tot, c_len = struct.unpack(
                            ">HHB", pkt_bytes[:5]
                        )
                        payload = bytes(pkt_bytes[5 : 5 + c_len])
                        chk = pkt_bytes[5 + c_len] if len(pkt_bytes) > 5 + c_len else -1

                        if (sum(payload) & 0xFF) == chk:
                            received_chunks[c_idx] = payload
                            total_expected_chunks = c_tot
                            recv_progress.max = c_tot
                            recv_progress.value = len(received_chunks)

                            receiver_status.value = (
                                f"<b>Status:</b> Decoded chunk {c_idx + 1}/{c_tot} ➔ Sending ACK beep..."
                            )
                            with out_receiver:
                                print(
                                    f" Decoded chunk {c_idx + 1:02d}/{c_tot:02d} ➔ Beeped ACK to Sender"
                                )

                            # Play ACK Confirmation Beep back to sender
                            time.sleep(0.05)
                            sd.play(
                                make_tone(FREQ_ACK, 0.18, 0.6),
                                samplerate=SAMPLE_RATE,
                                blocking=True,
                            )
                        else:
                            with out_receiver:
                                print(
                                    f"⚠️ Checksum error in chunk {c_idx + 1}. Skipping."
                                )

                    # Check if all chunks received
                    if (
                        total_expected_chunks
                        and len(received_chunks) == total_expected_chunks
                    ):
                        break

                    state = "SEARCH_CHUNK"
                    buffer = []  # Clear residual audio

                time.sleep(0.01)

        btn_recv_start.disabled = False
        btn_recv_cancel.disabled = True

        if rx_cancel_evt.is_set():
            receiver_status.value = (
                "<b style='color:#d9534f;'>Status: Reception CANCELLED by user.</b>"
            )
            with out_receiver:
                print("\n🛑 Reception cancelled.")
            return

        # Reconstruct full file
        if (
            total_expected_chunks
            and len(received_chunks) == total_expected_chunks
        ):
            try:
                comp_data = b"".join(
                    [received_chunks[i] for i in range(total_expected_chunks)]
                )
                decomp = gzip.decompress(comp_data)
                calc_hash = hashlib.sha256(decomp).hexdigest()

                save_p = txt_recv_file.value.strip()
                with open(save_p, "wb") as f:
                    f.write(decomp)

                receiver_status.value = (
                    "<b style='color:#5cb85c;'>Status: File Received & Verified!</b>"
                )
                with out_receiver:
                    print("\n" + "=" * 50)
                    print("🎉 FILE RECEIVED AND VERIFIED SUCCESSFULLY!")
                    print(f"💾 Saved to: {save_p}")
                    print(f"📊 Decompressed Size: {len(decomp):,} bytes")
                    print(f"🔑 SHA-256 Hash: {calc_hash}")
                    print("✅ Integrity: 100% (No packet loss)")
                    print("=" * 50)
            except Exception as e:
                with out_receiver:
                    print(f"\n❌ Reconstruction Error: {e}")
        else:
            with out_receiver:
                print(
                    f"\n⚠️ Incomplete reception: Received {len(received_chunks)}/{total_expected_chunks} chunks."
                )


def cancel_receiver(_):
    rx_cancel_evt.set()
    sd.stop()
    btn_recv_start.disabled = False
    btn_recv_cancel.disabled = True
