# ==============================================================================
# ROBUST DUAL-CRITERIA TONE DETECTOR (Absolute Volume + SNR + Duration)
# ==============================================================================


def detect_sustained_tone(
    audio_chunk,
    target_freq,
    tolerance=60,
    min_abs_peak=0.04,
    min_snr_ratio=4.0,
):
    """Returns True only if the target frequency exceeds BOTH absolute loudness and relative SNR thresholds."""
    if len(audio_chunk) < 512:
        return False

    # Check overall chunk amplitude first
    max_amplitude = np.max(np.abs(audio_chunk))
    if max_amplitude < min_abs_peak:
        return False  # Too quiet / ambient silence

    spectrum = np.abs(rfft(audio_chunk))
    freqs = rfftfreq(len(audio_chunk), 1.0 / SAMPLE_RATE)

    mask = (freqs >= target_freq - tolerance) & (freqs <= target_freq + tolerance)
    if not np.any(mask):
        return False

    target_peak = np.max(spectrum[mask])
    noise_floor = np.median(spectrum) + 1e-6
    snr = target_peak / noise_floor

    # Must pass both SNR and relative spectral dominance
    return snr >= min_snr_ratio

# UPGRADED TAB 2: BULLETPROOF PAIRING

pair_role = widgets.RadioButtons(
    options=["Laptop A (Initiator / Ping)", "Laptop B (Responder / Pong)"],
    description="This Device:",
    layout=widgets.Layout(width="380px"),
)
btn_pair_action = widgets.Button(
    description="Start Pairing Test", button_style="primary", icon="link"
)
btn_pair_cancel = widgets.Button(
    description="Stop", button_style="danger", icon="stop"
)
pair_status_banner = widgets.HTML(
    "<div style='padding:6px; background:#f0f0f0; border-radius:4px;'><b>Status:</b> Ready to pair.</div>"
)
tab2_out = widgets.Output()


def execute_pairing(_):
    pair_stop_evt.clear()
    btn_pair_action.disabled = True
    role = pair_role.value

    if "Initiator" in role:

        def init_worker():
            tab2_out.clear_output()
            with tab2_out:
                print("🔊 Sending Handshake Ping (1100 Hz)...")
            pair_status_banner.value = (
                "<div style='color:#0275d8;'><b>Status:</b> Ping sent! Waiting for Laptop B echo...</div>"
            )

            # Play Ping (0.4s)
            sd.play(
                make_tone(FREQ_PING, 0.4, 0.6),
                samplerate=SAMPLE_RATE,
                blocking=True,
            )

            # Wait 200ms to let local speaker sound completely die down
            time.sleep(0.2)

            consecutive_hits = 0
            heard_pong = False
            t0 = time.time()
            startup_frames_to_skip = 3  # Ignore mic open clicks

            def cb(indata, frames, t_info, status):
                nonlocal consecutive_hits, heard_pong, startup_frames_to_skip
                if pair_stop_evt.is_set():
                    raise sd.CallbackStop

                if startup_frames_to_skip > 0:
                    startup_frames_to_skip -= 1
                    return

                # Detect Pong (1350 Hz)
                if detect_sustained_tone(
                    indata[:, 0], FREQ_PONG, min_abs_peak=0.03, min_snr_ratio=4.0
                ):
                    consecutive_hits += 1
                    if consecutive_hits >= 2:  # Must persist for ~100ms
                        heard_pong = True
                else:
                    consecutive_hits = 0

            with sd.InputStream(
                channels=1,
                samplerate=SAMPLE_RATE,
                blocksize=2048,
                callback=cb,
            ):
                while (time.time() - t0 < 6.0) and not heard_pong:
                    if pair_stop_evt.is_set():
                        break
                    time.sleep(0.05)

            btn_pair_action.disabled = False
            tab2_out.clear_output()
            with tab2_out:
                if heard_pong:
                    pair_status_banner.value = (
                        "<div style='color:#5cb85c; font-weight:bold;'>✅ Status: PAIRED! Two-way audio verified.</div>"
                    )
                    print("🎉 Success! Two-way communication established.")
                else:
                    pair_status_banner.value = (
                        "<div style='color:#d9534f;'>❌ Status: No response from Laptop B.</div>"
                    )
                    print(
                        "Timed out. Ensure Laptop B is in 'Responder' mode and clicked 'Start Pairing Test' first."
                    )

        threading.Thread(target=init_worker).start()

    else:  # Responder

        def resp_worker():
            tab2_out.clear_output()
            with tab2_out:
                print(
                    "👂 Listening for Laptop A's ping... (Press Stop to cancel)"
                )
            pair_status_banner.value = (
                "<div style='color:#0275d8;'><b>Status:</b> Listening for Laptop A...</div>"
            )

            consecutive_hits = 0
            heard_ping = False
            startup_frames_to_skip = 3  # Ignore mic initialization noise

            def cb(indata, frames, t_info, status):
                nonlocal consecutive_hits, heard_ping, startup_frames_to_skip
                if pair_stop_evt.is_set():
                    raise sd.CallbackStop

                if startup_frames_to_skip > 0:
                    startup_frames_to_skip -= 1
                    return

                # Detect Ping (1100 Hz)
                if detect_sustained_tone(
                    indata[:, 0], FREQ_PING, min_abs_peak=0.03, min_snr_ratio=4.0
                ):
                    consecutive_hits += 1
                    if consecutive_hits >= 2:  # Must persist for ~100ms
                        heard_ping = True
                else:
                    consecutive_hits = 0

            with sd.InputStream(
                channels=1,
                samplerate=SAMPLE_RATE,
                blocksize=2048,
                callback=cb,
            ):
                while not heard_ping and not pair_stop_evt.is_set():
                    time.sleep(0.04)

            if heard_ping and not pair_stop_evt.is_set():
                time.sleep(0.15)  # Wait for incoming ping tone to finish
                # Reply with Pong (1350 Hz)
                sd.play(
                    make_tone(FREQ_PONG, 0.4, 0.6),
                    samplerate=SAMPLE_RATE,
                    blocking=True,
                )
                pair_status_banner.value = (
                    "<div style='color:#5cb85c; font-weight:bold;'>✅ Status: Heard Ping & Replied with Pong! Paired.</div>"
                )
                tab2_out.clear_output()
                with tab2_out:
                    print("⚡ Genuine Ping detected! Sent Pong back.")

            btn_pair_action.disabled = False

        threading.Thread(target=resp_worker).start()


def cancel_pairing(_):
    pair_stop_evt.set()
    sd.stop()
    btn_pair_action.disabled = False
    pair_status_banner.value = "<b>Status:</b> Pairing stopped."
    tab2_out.clear_output()
    with tab2_out:
        print("🛑 Pairing cancelled.")


btn_pair_action.on_click(execute_pairing)
btn_pair_cancel.on_click(cancel_pairing)

tab_pair = widgets.VBox(
    [
        widgets.HTML("<h4>Step 2: Streamlined Device Pairing</h4>"),
        widgets.HTML(
            "<p><b>How to pair:</b><br>"
            "1. On Laptop B: Select <b>'Responder'</b> and click <b>Start Pairing Test</b> (it will wait quietly).<br>"
            "2. On Laptop A: Select <b>'Initiator'</b> and click <b>Start Pairing Test</b>.</p>"
        ),
        pair_role,
        widgets.HBox([btn_pair_action, btn_pair_cancel]),
        pair_status_banner,
        tab2_out,
    ]
)
