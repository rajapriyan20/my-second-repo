import base64
import gzip
import hashlib
import os
import struct
import threading
import time
import ipywidgets as widgets
from IPython.display import display
import numpy as np
from scipy.fft import rfft, rfftfreq
import sounddevice as sd

# --- Acoustic Protocol Parameters ---
SAMPLE_RATE = 44100
SYMBOL_DURATION = 0.025  # 25 ms per byte (~40 bytes/sec raw)
SAMPLES_PER_SYMBOL = int(SAMPLE_RATE * SYMBOL_DURATION)

# Dual-Band 16-FSK Frequencies (Low Nibble + High Nibble)
LOW_FREQS = [1200 + i * 120 for i in range(16)]  # 1200 Hz to 3000 Hz
HIGH_FREQS = [3400 + i * 120 for i in range(16)]  # 3400 Hz to 5200 Hz

# Control Frequencies (Handshake & Sync)
TONE_PING = 800
TONE_PONG = 950
TONE_PREAMBLE = 6000

# --- Modulation (Transmitter) ---


def generate_tone(freq, duration):
    t = np.linspace(0, duration, int(SAMPLE_RATE * duration), endpoint=False)
    # Apply cosine envelope to prevent audio clicking
    envelope = np.sin(np.pi * np.linspace(0, 1, len(t)))
    return (np.sin(2 * np.pi * freq * t) * envelope * 0.5).astype(np.float32)


def encode_byte_stream(data_bytes):
    """Encodes a byte array into a continuous dual-tone audio wave."""
    audio_segments = []

    # 1. Preamble sync chirp (300 ms)
    audio_segments.append(generate_tone(TONE_PREAMBLE, 0.3))
    audio_segments.append(
        np.zeros(int(SAMPLE_RATE * 0.05), dtype=np.float32)
    )  # 50ms guard

    # 2. Encode each byte as a pair of simultaneous frequencies
    t = np.linspace(
        0, SYMBOL_DURATION, SAMPLES_PER_SYMBOL, endpoint=False
    ).astype(np.float32)
    envelope = np.sin(np.pi * np.linspace(0, 1, SAMPLES_PER_SYMBOL)).astype(
        np.float32
    )

    for b in data_bytes:
        low_nibble = b & 0x0F
        high_nibble = (b >> 4) & 0x0F

        f1 = LOW_FREQS[low_nibble]
        f2 = HIGH_FREQS[high_nibble]

        tone = (np.sin(2 * np.pi * f1 * t) + np.sin(2 * np.pi * f2 * t)) * 0.25
        audio_segments.append((tone * envelope).astype(np.float32))

    # 3. Postamble sync
    audio_segments.append(generate_tone(TONE_PREAMBLE, 0.2))
    return np.concatenate(audio_segments)


# --- Demodulation (Receiver) ---


def detect_frequency(
    audio_block, target_freqs, tolerance=60, min_threshold=0.08
):
    """Detects the strongest matching target frequency using FFT."""
    if len(audio_block) == 0:
        return None
    spectrum = np.abs(rfft(audio_block))
    freqs = rfftfreq(len(audio_block), 1.0 / SAMPLE_RATE)

    max_mag = -1
    best_idx = None

    for idx, target in enumerate(target_freqs):
        mask = (freqs >= target - tolerance) & (freqs <= target + tolerance)
        if np.any(mask):
            mag = np.max(spectrum[mask])
            if mag > max_mag and mag > min_threshold:
                max_mag = mag
                best_idx = idx

    return best_idx


# --- Jupyter GUI & Controller ---

mode_selector = widgets.ToggleButtons(
    options=["Sender", "Receiver"], description="Role:", button_style="primary"
)
dynamic_container = widgets.VBox([])
active_stop_event = None


def render_ui(mode):
    global active_stop_event
    if active_stop_event:
        active_stop_event.set()

    if mode == "Sender":
        file_input = widgets.Text(
            value="data.txt",
            description="File Path:",
            layout=widgets.Layout(width="450px"),
        )
        btn_check = widgets.Button(
            description="Check Receiver Device",
            button_style="info",
            icon="plug",
        )
        btn_send = widgets.Button(
            description="Start Sending",
            button_style="success",
            icon="paper-plane",
        )
        status_box = widgets.Output()

        def on_check_receiver_clicked(_):
            status_box.clear_output()
            with status_box:
                print("🔊 Sending Handshake Tone (Ping)...")

            def worker():
                # Play Ping Tone
                ping_wave = generate_tone(TONE_PING, 0.4)
                sd.play(ping_wave, samplerate=SAMPLE_RATE, blocking=True)

                # Listen for Pong response
                with status_box:
                    print("👂 Waiting for Receiver ACK...")

                ack_heard = False
                start_t = time.time()

                def record_callback(indata, frames, t_info, status):
                    nonlocal ack_heard
                    blk = indata[:, 0]
                    if (
                        detect_frequency(
                            blk, [TONE_PONG], tolerance=50, min_threshold=0.1
                        )
                        is not None
                    ):
                        ack_heard = True

                with sd.InputStream(
                    channels=1,
                    samplerate=SAMPLE_RATE,
                    blocksize=2048,
                    callback=record_callback,
                ):
                    while time.time() - start_t < 6.0 and not ack_heard:
                        time.sleep(0.05)

                with status_box:
                    if ack_heard:
                        print("✅ Handshake Verified! Receiver is online.")
                    else:
                        print("❌ No response. Check volume and proximity.")

            threading.Thread(target=worker).start()

        def on_send_clicked(_):
            status_box.clear_output()
            path = file_input.value.strip()

            if not os.path.exists(path):
                with status_box:
                    print(f"❌ Error: File '{path}' not found.")
                return

            with open(path, "rb") as f:
                raw_bytes = f.read()

            file_hash = hashlib.sha256(raw_bytes).hexdigest()
            compressed = gzip.compress(raw_bytes)

            # Frame Structure: [4-byte Compressed Length] + [Compressed Data] + [32-byte SHA256 Hash]
            payload = (
                struct.pack(">I", len(compressed))
                + compressed
                + file_hash.encode("ascii")
            )
            wave = encode_byte_stream(payload)
            est_time = len(wave) / SAMPLE_RATE

            with status_box:
                print(f"📦 File: {os.path.basename(path)}")
                print(f"📊 Original Size: {len(raw_bytes):,} bytes")
                print(f"🗜️ Compressed Size: {len(compressed):,} bytes")
                print(f"🔑 SHA-256: {file_hash}")
                print(
                    f"🚀 Transmitting (~{est_time:.1f}s)... Keep room quiet."
                )

            def play_worker():
                sd.play(wave, samplerate=SAMPLE_RATE, blocking=True)
                with status_box:
                    print("\n✨ Sound transmission finished!")

            threading.Thread(target=play_worker).start()

        btn_check.on_click(on_check_receiver_clicked)
        btn_send.on_click(on_send_clicked)

        dynamic_container.children = [
            file_input,
            widgets.HBox([btn_check, btn_send]),
            status_box,
        ]

    else:  # Receiver Mode
        save_input = widgets.Text(
            value="received_file.txt",
            description="Save As:",
            layout=widgets.Layout(width="450px"),
        )
        btn_check = widgets.Button(
            description="Check Sender Device",
            button_style="info",
            icon="plug",
        )
        btn_receive = widgets.Button(
            description="Start Receiving",
            button_style="success",
            icon="headphones",
        )
        btn_stop = widgets.Button(
            description="Stop", button_style="danger", icon="stop"
        )
        status_box = widgets.Output()

        def on_check_sender_clicked(_):
            status_box.clear_output()
            with status_box:
                print("👂 Listening for Sender Ping (8s timeout)...")

            def worker():
                heard_ping = False
                start_t = time.time()

                def callback(indata, frames, t_info, status):
                    nonlocal heard_ping
                    if (
                        detect_frequency(
                            indata[:, 0],
                            [TONE_PING],
                            tolerance=50,
                            min_threshold=0.1,
                        )
                        is not None
                    ):
                        heard_ping = True

                with sd.InputStream(
                    channels=1,
                    samplerate=SAMPLE_RATE,
                    blocksize=2048,
                    callback=callback,
                ):
                    while time.time() - start_t < 8.0 and not heard_ping:
                        time.sleep(0.05)

                if heard_ping:
                    with status_box:
                        print("📡 Ping detected! Responding with Pong...")
                    pong_wave = generate_tone(TONE_PONG, 0.4)
                    sd.play(pong_wave, samplerate=SAMPLE_RATE, blocking=True)
                    with status_box:
                        print("✅ Handshake complete.")
                else:
                    with status_box:
                        print("❌ Timed out waiting for Ping.")

            threading.Thread(target=worker).start()

        def on_receive_clicked(_):
            global active_stop_event
            status_box.clear_output()
            active_stop_event = threading.Event()
            stop_evt = active_stop_event

            with status_box:
                print(
                    "🎧 Listening for incoming transmission... (Click 'Start Sending' on Sender)"
                )

            def rx_worker():
                buffer = []
                state = "WAIT_PREAMBLE"
                received_bytes = bytearray()
                expected_len = None
                expected_hash = None

                def audio_cb(indata, frames, t_info, status):
                    if stop_evt.is_set():
                        raise sd.CallbackStop
                    buffer.extend(indata[:, 0])

                with sd.InputStream(
                    channels=1,
                    samplerate=SAMPLE_RATE,
                    blocksize=1024,
                    callback=audio_cb,
                ):
                    while not stop_evt.is_set():
                        if state == "WAIT_PREAMBLE":
                            if len(buffer) >= 4096:
                                chunk = np.array(buffer[:4096])
                                if (
                                    detect_frequency(
                                        chunk,
                                        [TONE_PREAMBLE],
                                        tolerance=100,
                                        min_threshold=0.15,
                                    )
                                    is not None
                                ):
                                    state = "PREAMBLE_FOUND"
                                    # Skip ahead past preamble
                                    buffer = buffer[
                                        int(
                                            4096
                                            + SAMPLE_RATE
                                            * (0.3 + 0.05 - 0.05)
                                        ) :
                                    ]
                                    with status_box:
                                        print("⚡ Signal detected! Decoding...")
                                else:
                                    buffer = buffer[2048:]

                        elif state == "PREAMBLE_FOUND":
                            # Process symbol by symbol
                            while len(buffer) >= SAMPLES_PER_SYMBOL:
                                sym = np.array(buffer[:SAMPLES_PER_SYMBOL])
                                buffer = buffer[SAMPLES_PER_SYMBOL:]

                                # Check for end preamble
                                if (
                                    detect_frequency(
                                        sym,
                                        [TONE_PREAMBLE],
                                        tolerance=100,
                                        min_threshold=0.15,
                                    )
                                    is not None
                                ):
                                    state = "DONE"
                                    stop_evt.set()
                                    break

                                l_idx = detect_frequency(
                                    sym, LOW_FREQS, tolerance=50
                                )
                                h_idx = detect_frequency(
                                    sym, HIGH_FREQS, tolerance=50
                                )

                                if l_idx is not None and h_idx is not None:
                                    val = (h_idx << 4) | l_idx
                                    received_bytes.append(val)

                                    # Check if we have received the 4-byte header
                                    if (
                                        len(received_bytes) == 4
                                        and expected_len is None
                                    ):
                                        expected_len = struct.unpack(
                                            ">I", received_bytes[:4]
                                        )[0]

                        time.sleep(0.02)

                # Process Decoded Data
                if len(received_bytes) > 4 + 64:
                    try:
                        comp_len = struct.unpack(">I", received_bytes[:4])[0]
                        comp_data = bytes(received_bytes[4 : 4 + comp_len])
                        exp_hash = bytes(
                            received_bytes[4 + comp_len : 4 + comp_len + 64]
                        ).decode("ascii")

                        decomp = gzip.decompress(comp_data)
                        calc_hash = hashlib.sha256(decomp).hexdigest()

                        save_path = save_input.value.strip()
                        with open(save_path, "wb") as f:
                            f.write(decomp)

                        with status_box:
                            print("\n" + "=" * 45)
                            print("🎉 File Received & Decoded Successfully!")
                            print(f"💾 Saved to: {save_path}")
                            print(f"🔑 Expected Hash: {exp_hash}")
                            print(f"🔑 Received Hash: {calc_hash}")
                            if calc_hash == exp_hash:
                                print(
                                    "✅ Integrity Check: 100% MATCH (No errors)"
                                )
                            else:
                                print(
                                    "❌ Integrity Check: HASH MISMATCH (Noise corruption)"
                                )
                            print("=" * 45)
                    except Exception as e:
                        with status_box:
                            print(f"\n❌ Reconstruction Error: {e}")
                else:
                    with status_box:
                        print(
                            f"\n⚠️ Incomplete signal: Captured {len(received_bytes)} bytes."
                        )

            threading.Thread(target=rx_worker).start()

        def on_stop_clicked(_):
            global active_stop_event
            if active_stop_event:
                active_stop_event.set()
            with status_box:
                print("\n🛑 Listening cancelled.")

        btn_check.on_click(on_check_sender_clicked)
        btn_receive.on_click(on_receive_clicked)
        btn_stop.on_click(on_stop_clicked)

        dynamic_container.children = [
            save_input,
            widgets.HBox([btn_check, btn_receive, btn_stop]),
            status_box,
        ]


def on_mode_change(change):
    render_ui(change["new"])


mode_selector.observe(on_mode_change, names="value")

# Initial Render
display(mode_selector)
display(dynamic_container)
render_ui("Sender")
