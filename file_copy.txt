import gzip
import hashlib
import os
import struct
import threading
import time
import ipywidgets as widgets
from IPython.display import clear_output, display
import numpy as np
from scipy.fft import rfft, rfftfreq
import sounddevice as sd

# --- Audio & Physical Layer Settings ---
SAMPLE_RATE = 44100
SYMBOL_DURATION = 0.035  # 35ms per symbol (resilient to room reverb)
SAMPLES_PER_SYM = int(SAMPLE_RATE * SYMBOL_DURATION)

# Frequencies for 16-FSK (Hex nibbles 0x0 to 0xF)
# Spaced across 1600 Hz - 4000 Hz (Optimal for laptop mic/speaker response)
FSK_FREQS = [1600 + i * 160 for i in range(16)]

# Special Control / Handshake Frequencies
FREQ_PING = 1100
FREQ_PONG = 1350
FREQ_PREAMBLE = 4800

# --- Helper Signal Generation ---


def make_tone(freq, duration=0.4, amp=0.5):
    """Generates a smoothed sine wave tone."""
    t = np.linspace(0, duration, int(SAMPLE_RATE * duration), endpoint=False)
    # Cosine window to eliminate speaker clicks
    window = np.sin(np.pi * np.linspace(0, 1, len(t)))
    return (np.sin(2 * np.pi * freq * t) * window * amp).astype(np.float32)


def make_chirp(start_f, end_f, duration=0.25, amp=0.5):
    """Generates a linear frequency sweep for synchronization."""
    t = np.linspace(0, duration, int(SAMPLE_RATE * duration), endpoint=False)
    phase = 2 * np.pi * (start_f * t + (end_f - start_f) * (t**2) / (2 * duration))
    window = np.sin(np.pi * np.linspace(0, 1, len(t)))
    return (np.sin(phase) * window * amp).astype(np.float32)


def get_peak_snr(audio_chunk, target_freq, tolerance=60):
    """Calculates Signal-to-Noise Ratio (SNR) for a specific frequency target."""
    if len(audio_chunk) < 512:
        return 0.0
    spectrum = np.abs(rfft(audio_chunk))
    freqs = rfftfreq(len(audio_chunk), 1.0 / SAMPLE_RATE)

    target_mask = (freqs >= target_freq - tolerance) & (
        freqs <= target_freq + tolerance
    )
    if not np.any(target_mask):
        return 0.0

    target_power = np.max(spectrum[target_mask])
    noise_floor = np.median(spectrum) + 1e-6
    return target_power / noise_floor


def decode_fsk_nibble(audio_chunk):
    """Identifies the dominant FSK frequency bin using relative energy."""
    spectrum = np.abs(rfft(audio_chunk))
    freqs = rfftfreq(len(audio_chunk), 1.0 / SAMPLE_RATE)

    max_snr = 0
    best_nibble = None

    noise_floor = np.median(spectrum) + 1e-6

    for nibble, freq in enumerate(FSK_FREQS):
        mask = (freqs >= freq - 50) & (freqs <= freq + 50)
        if np.any(mask):
            peak = np.max(spectrum[mask])
            snr = peak / noise_floor
            if snr > max_snr and snr > 3.0:  # 3x noise floor threshold
                max_snr = snr
                best_nibble = nibble

    return best_nibble

    max=1.0,

def run_speaker_test(_):
    with diag_out:
        clear_output()
        print("🔊 Playing melodic test tone (Low -> High sweep)...")

    # Generate a smooth arpeggio
    notes = [440, 554, 659, 880]
    audio = np.concatenate([make_tone(n, 0.15, 0.4) for n in notes])
    sd.play(audio, samplerate=SAMPLE_RATE, blocking=False)


def run_mic_test(_):
    btn_mic_test.disabled = True

    def worker():
        with diag_out:
            clear_output()
            print("🎙️ Listening... Please speak or tap near the laptop.")

        peak_levels = []

        def audio_cb(indata, frames, t_info, status):
            level = float(np.max(np.abs(indata[:, 0])))
            peak_levels.append(level)
            mic_progress.value = min(level * 3.0, 1.0)  # Visual amplifier

        stream = sd.InputStream(
            channels=1, samplerate=SAMPLE_RATE, blocksize=2048, callback=audio_cb
        )
        with stream:
            time.sleep(5.0)

        btn_mic_test.disabled = False
        mic_progress.value = 0
        avg_peak = np.mean(peak_levels) if peak_levels else 0

        with diag_out:
            clear_output()
            if avg_peak > 0.02:
                print(
                    f"✅ Microphone Working! Detected average level: {avg_peak * 100:.1f}%"
                )
            else:
                print(
                    "⚠️ Very low input detected. Please increase your microphone input gain in OS settings."
                )

    threading.Thread(target=worker).start()

pair_out = widgets.Output()
btn_send_ping = widgets.Button(
    description="Send Ping (Tester)", button_style="primary", icon="podcast"
)
btn_auto_beacon = widgets.ToggleButton(
    value=False,
    description="Echo Responder (Receiver)",
    button_style="info",
    icon="satellite-dish",
)
pair_status = widgets.Label(value="Status: Ready")

beacon_stop = threading.Event()


def send_ping_action(_):
    btn_send_ping.disabled = True
    pair_status.value = "Status: Sending Ping & Waiting for Echo..."

    def worker():
        with pair_out:
            clear_output()
            print("🔊 Transmitting Ping tone (1100 Hz)...")

        # Play Ping
        sd.play(make_tone(FREQ_PING, 0.4, 0.6), samplerate=SAMPLE_RATE, blocking=True)

        # Listen for Pong Echo
        heard_pong = False
        t_start = time.time()

        def listen_cb(indata, frames, t_info, status):
            nonlocal heard_pong
            snr = get_peak_snr(indata[:, 0], FREQ_PONG)
            if snr > 4.0:
                heard_pong = True

        stream = sd.InputStream(
            channels=1,
            samplerate=SAMPLE_RATE,
            blocksize=2048,
            callback=listen_cb,
        )
        with stream:
            while (time.time() - t_start < 6.0) and not heard_pong:
                time.sleep(0.05)

        btn_send_ping.disabled = False
        with pair_out:
            clear_output()
            if heard_pong:
                pair_status.value = "Status: Connected ✅"
                print("🎉 SUCCESS! Received echo response from the other laptop.")
                print(
                    "Signal path verified: Speakers and Microphones are calibrated."
                )
            else:
                pair_status.value = "Status: No response ❌"
                print(
                    "❌ No echo received within 6s. Make sure the other laptop enabled 'Echo Responder'."
                )

    threading.Thread(target=worker).start()


def toggle_beacon(change):
    if change["new"]:
        beacon_stop.clear()
        btn_auto_beacon.button_style = "danger"
        btn_auto_beacon.description = "Stop Responder"
        pair_status.value = "Status: Active & Listening for Pings..."

        def worker():
            with pair_out:
                clear_output()
                print(
                    "📡 Auto-responder active. Waiting to hear Ping from the sender..."
                )

            def cb(indata, frames, t_info, status):
                snr = get_peak_snr(indata[:, 0], FREQ_PING)
                if snr > 4.0:
                    # Heard ping -> reply with pong!
                    time.sleep(0.1)
                    sd.play(
                        make_tone(FREQ_PONG, 0.4, 0.6),
                        samplerate=SAMPLE_RATE,
                        blocking=False,
                    )
                    with pair_out:
                        print("⚡ Heard Ping! Sent Pong back.")

            stream = sd.InputStream(
                channels=1, samplerate=SAMPLE_RATE, blocksize=2048, callback=cb
            )
            with stream:
                while not beacon_stop.is_set():
                    time.sleep(0.1)

            with pair_out:
                print("🛑 Responder stopped.")

        threading.Thread(target=worker).start()
    else:
        beacon_stop.set()
        btn_auto_beacon.button_style = "info"
        btn_auto_beacon.description = "Echo Responder (Receiver)"
        pair_status.value = "Status: Idle"


btn_send_ping.on_click(send_ping_action)
btn_auto_beacon.observe(toggle_beacon, names="value")

# --- SENDER WIDGETS ---
txt_file_path = widgets.Text(
    value="data.txt",
    description="File:",
    layout=widgets.Layout(width="350px"),
)
btn_send_file = widgets.Button(
    description="Transmit File", button_style="success", icon="paper-plane"
)
send_progress = widgets.IntProgress(
    value=0,
    min=0,
    max=100,
    description="Sending:",
    bar_style="info",
    layout=widgets.Layout(width="400px"),
)
sender_out = widgets.Output()

# --- RECEIVER WIDGETS ---
txt_save_path = widgets.Text(
    value="received_file.txt",
    description="Save As:",
    layout=widgets.Layout(width="350px"),
)
btn_start_recv = widgets.Button(
    description="Start Listening", button_style="success", icon="headphones"
)

    description="Cancel", button_style="danger", icon="stop"
)
recv_progress = widgets.IntProgress(
    value=0,
    min=0,
    max=100,
    description="Receiving:",
    bar_style="info",
    layout=widgets.Layout(width="400px"),
)
receiver_out = widgets.Output()

rx_stop_event = threading.Event()


def execute_file_transmission(_):
    path = txt_file_path.value.strip()
    if not os.path.exists(path):
        with sender_out:
            clear_output()
            print(f"❌ Error: File '{path}' does not exist.")
        return

    with open(path, "rb") as f:
        raw_bytes = f.read()

    file_hash = hashlib.sha256(raw_bytes).hexdigest()
    compressed = gzip.compress(raw_bytes)

    # Packet framing: [4 bytes length] + [compressed payload] + [32 bytes SHA256 hex]
    payload = (
        struct.pack(">I", len(compressed))
        + compressed
        + file_hash[:16].encode("ascii")
    )
    total_nibbles = len(payload) * 2

    # Build audio stream
    audio_blocks = []
    # 1. Preamble Chirp
    audio_blocks.append(make_chirp(2000, 4800, duration=0.4, amp=0.6))
    audio_blocks.append(
        np.zeros(int(SAMPLE_RATE * 0.05), dtype=np.float32)
    )  # Guard interval

    t_sym = np.linspace(
        0, SYMBOL_DURATION, SAMPLES_PER_SYM, endpoint=False
    ).astype(np.float32)
    sym_win = np.sin(np.pi * np.linspace(0, 1, SAMPLES_PER_SYM)).astype(
        np.float32
    )

    for byte in payload:
        # High nibble then low nibble
        for nibble in [(byte >> 4) & 0x0F, byte & 0x0F]:
            freq = FSK_FREQS[nibble]
            tone = (np.sin(2 * np.pi * freq * t_sym) * sym_win * 0.5).astype(
                np.float32
            )
            audio_blocks.append(tone)

    # Postamble Chirp
    audio_blocks.append(make_chirp(4800, 2000, duration=0.3, amp=0.6))
    full_audio = np.concatenate(audio_blocks)

    duration_sec = len(full_audio) / SAMPLE_RATE

    with sender_out:
        clear_output()
        print(f"📦 File: {os.path.basename(path)}")
        print(
            f"📊 Size: {len(raw_bytes):,} bytes  ➔  Compressed: {len(compressed):,} bytes"
        )
        print(f"🔑 SHA-256 (Prefix): {file_hash[:16]}")
        print(
            f"⏳ Est. Time: {duration_sec:.1f}s across {total_nibbles} audio symbols"
        )

    def tx_worker():
        btn_send_file.disabled = True
        send_progress.max = total_nibbles
        send_progress.value = 0

        sd.play(full_audio, samplerate=SAMPLE_RATE, blocking=False)

        # Update progress bar smoothly with elapsed time
        start_time = time.time()
        while time.time() - start_time < duration_sec:
            elapsed = time.time() - start_time
            curr_sym = int((elapsed / duration_sec) * total_nibbles)
            send_progress.value = min(curr_sym, total_nibbles)
            time.sleep(0.1)

        send_progress.value = total_nibbles
        btn_send_file.disabled = False
        with sender_out:
            print("\n✨ Transmission Complete!")

    threading.Thread(target=tx_worker).start()


def execute_file_reception(_):
    rx_stop_event.clear()
    btn_start_recv.disabled = True
    recv_progress.value = 0

    with receiver_out:
        clear_output()
        print(
            "🎧 Listening for incoming transmission chirp... (Press 'Cancel' to abort)"
        )

    def rx_worker():
        buffer = []
        state = "WAIT_PREAMBLE"
        received_nibbles = []
        expected_len = None

        def audio_cb(indata, frames, t_info, status):
            if rx_stop_event.is_set():
                raise sd.CallbackStop
            buffer.extend(indata[:, 0])

        stream = sd.InputStream(
            channels=1,
            samplerate=SAMPLE_RATE,
            blocksize=1024,
            callback=audio_cb,
        )
        with stream:
            while not rx_stop_event.is_set():
                if state == "WAIT_PREAMBLE":
                    if len(buffer) >= 8192:
                        chunk = np.array(buffer[:8192])
                        snr = get_peak_snr(chunk, FREQ_PREAMBLE, tolerance=150)
                        if snr > 3.5:
                            state = "READING_DATA"
                            # Advance past the preamble
                            buffer = buffer[
                                int(
                                    4096
                                    + SAMPLE_RATE
                                    * (SYMBOL_DURATION + 0.05)
                                ) :
                            ]
                            with receiver_out:
                                print(
                                    "⚡ Transmission detected! Decoding data stream..."
                                )
                        else:
                            buffer = buffer[2048:]

                elif state == "READING_DATA":
                    while len(buffer) >= SAMPLES_PER_SYM:
                        sym_chunk = np.array(buffer[:SAMPLES_PER_SYM])
                        buffer = buffer[SAMPLES_PER_SYM:]

                        nibble = decode_fsk_nibble(sym_chunk)
                        if nibble is not None:
                            received_nibbles.append(nibble)
                            recv_progress.value = len(received_nibbles)

                            # Determine total length once 4-byte header (8 nibbles) is received
                            if (
                                len(received_nibbles) == 8
                                and expected_len is None
                            ):
                                header_bytes = bytearray()
                                for i in range(0, 8, 2):
                                    header_bytes.append(
                                        (received_nibbles[i] << 4)
                                        | received_nibbles[i + 1]
                                    )
                                comp_len = struct.unpack(
                                    ">I", header_bytes[:4]
                                )[0]
                                expected_len = comp_len
                                total_expected_nibbles = (
                                    4 + comp_len + 16
                                ) * 2
                                recv_progress.max = total_expected_nibbles
                                with receiver_out:
                                    print(
                                        f"📥 Incoming payload: {comp_len:,} compressed bytes"
                                    )

                            # Check completion
                            if (
                                expected_len
                                and len(received_nibbles)
                                >= (4 + expected_len + 16) * 2
                            ):
                                state = "DONE"
                                rx_stop_event.set()
                                break

                time.sleep(0.01)

        btn_start_recv.disabled = False

        # Reconstruct byte array
        if len(received_nibbles) >= 8:
            raw_bytes = bytearray()
            for i in range(0, len(received_nibbles) - 1, 2):
                raw_bytes.append(
                    (received_nibbles[i] << 4) | received_nibbles[i + 1]
                )

            try:
                comp_len = struct.unpack(">I", raw_bytes[:4])[0]
                comp_data = bytes(raw_bytes[4 : 4 + comp_len])
                exp_hash_prefix = bytes(
                    raw_bytes[4 + comp_len : 4 + comp_len + 16]
                ).decode("ascii")

                decomp = gzip.decompress(comp_data)
                calc_hash = hashlib.sha256(decomp).hexdigest()

                save_path = txt_save_path.value.strip()
                with open(save_path, "wb") as f:
                    f.write(decomp)

                with receiver_out:
                    print("\n" + "=" * 46)
                    print("🎉 File Received and Verified!")
                    print(f"💾 Saved to: {save_path}")
                    print(f"📊 Decompressed Size: {len(decomp):,} bytes")
                    print(f"🔑 Expected Hash (Prefix): {exp_hash_prefix}")
                    print(f"🔑 Received Hash (Full):   {calc_hash}")
                    if calc_hash.startswith(exp_hash_prefix):
                        print("✅ Integrity Check: 100% MATCH")
                    else:
                        print(
                            "❌ Integrity Check: HASH MISMATCH (Noise corruption detected)"
                        )
                    print("=" * 46)
            except Exception as e:
                with receiver_out:
                    print(
                        f"\n⚠️ Decoding Error: {e}. Ensure volume is clear and retry."
                    )
        else:
            with receiver_out:
                print("\n🛑 Reception ended with incomplete or empty signal.")

    threading.Thread(target=rx_worker).start()


def cancel_file_reception(_):
    rx_stop_event.set()
    btn_start_recv.disabled = False
    with receiver_out:
        print("\n🛑 Reception cancelled.")


btn_send_file.on_click(execute_file_transmission)

btn_stop_recv.on_click(cancel_file_reception)
